"""
TrendZone AI OS
================================

Base Agent.

Classe base para todos os agentes
do sistema.

Todos os Agents deverÃ£o herdar
desta classe.

Nunca implemente um Agent
sem utilizar BaseAgent.
"""

from abc import ABC
from abc import abstractmethod
from time import perf_counter
from uuid import uuid4

from app.core.logger import TZLogger


class BaseAgent(ABC):

    NAME = "BASE_AGENT"

    VERSION = "1.0"

    DESCRIPTION = ""

    def __init__(self):

        self.id = str(uuid4())

        self.execution_time = 0

        self.initialized = False

        self.finished = False

    # =====================================================
    # LIFECYCLE
    # =====================================================

    def initialize(self):

        self.initialized = True

        TZLogger.info(

            self.NAME,

            "Inicializado."

        )

    def validate(self):

        return True

    def before_execute(self):

        pass

    @abstractmethod
    def execute(self, context):

        """
        MÃ©todo obrigatÃ³rio.
        """

    def after_execute(self):

        pass

    def finish(self):

        self.finished = True

        TZLogger.success(

            self.NAME,

            "Finalizado."

        )

    # =====================================================
    # EXECUÃ‡ÃƒO
    # =====================================================

    def run(self, context=None):

        start = perf_counter()

        self.initialize()

        self.validate()

        self.before_execute()

        result = self.execute(context)

        self.after_execute()

        self.execution_time = (

            perf_counter() - start

        )

        self.finish()

        return result

    # =====================================================
    # STATUS
    # =====================================================

    def status(self):

        return {

            "id": self.id,

            "name": self.NAME,

            "version": self.VERSION,

            "initialized": self.initialized,

            "finished": self.finished,

            "execution_time": round(

                self.execution_time,

                4,

            ),

        }

"""

Objeto compartilhado entre todos

Todo Agent recebe um AgentContext.

from datetime import datetime
from uuid import uuid4


class AgentContext:

    def __init__(self):

        self.execution_id = str(uuid4())

        self.started_at = datetime.utcnow()

        self.finished_at = None

        self.current_agent = None

        self.products = []

        self.reports = []

        self.decisions = []

        self.metrics = {}

        self.metadata = {}

        self.errors = []

    # =====================================================
    # PRODUCTS
    # =====================================================

    def set_products(self, products):

        self.products = products

    def get_products(self):

        return self.products

    # =====================================================
    # REPORTS
    # =====================================================

    def add_report(self, report):

        self.reports.append(report)

    def set_reports(self, reports):

        self.reports = reports

    def get_reports(self):

        return self.reports

    # =====================================================
    # DECISIONS
    # =====================================================

    def add_decision(self, decision):

        self.decisions.append(decision)

    def get_decisions(self):

        return self.decisions

    # =====================================================
    # METRICS
    # =====================================================

    def set_metric(self, key, value):

        self.metrics[key] = value

    def get_metric(self, key):

        return self.metrics.get(key)

    # =====================================================
    # METADATA
    # =====================================================

    def set_metadata(self, key, value):

        self.metadata[key] = value

    def get_metadata(self, key):

        return self.metadata.get(key)

    # =====================================================
    # ERRORS
    # =====================================================

    def add_error(self, error):

        self.errors.append(error)

    # =====================================================
    # EXECUTION
    # =====================================================

    def finish(self):

        self.finished_at = datetime.utcnow()

    @property
    def duration(self):

        if not self.finished_at:

            return None

        return round(

            (

                self.finished_at

                - self.started_at

            ).total_seconds(),

            4,

        )

        return agents

"""
TrendZone AI OS
================================

Pipeline.

Define a sequÃªncia oficial
de execuÃ§Ã£o dos Agents.

O Orchestrator nunca deverÃ¡
conhecer diretamente os Agents.

Ele apenas executa este Pipeline.
"""

from dataclasses import dataclass, field


@dataclass(slots=True)
class PipelineStep:

    agent_id: str

    enabled: bool = True


class Pipeline:

    def __init__(self):

        self.steps: list[PipelineStep] = []

    # =====================================================
    # CONFIGURAÃ‡ÃƒO
    # =====================================================

    def add(self, agent_id: str):

        self.steps.append(

            PipelineStep(agent_id)

        )

        return self

    def disable(self, agent_id: str):

        for step in self.steps:

            if step.agent_id == agent_id:

                step.enabled = False

    def enable(self, agent_id: str):

        for step in self.steps:

            if step.agent_id == agent_id:

                step.enabled = True

    # =====================================================
    # CONSULTAS
    # =====================================================

    def get_steps(self):

        return [

            step

            for step in self.steps

            if step.enabled

        ]

    def names(self):

        return [

            step.agent_id

            for step in self.get_steps()

        ]

    def clear(self):

        self.steps.clear()

    def __len__(self):

        return len(self.steps)

"""

Global Middleware

ResponsÃ¡vel por:

- Logging

- Request ID
- Headers
- Auditoria

Arquitetura Enterprise.
"""

from time import perf_counter
from uuid import uuid4

from fastapi import Request

from app.core.logger import TZLogger


class RequestMiddleware:

    @staticmethod
    async def dispatch(request: Request, call_next):

        request_id = str(uuid4())

        start = perf_counter()

        TZLogger.info(

            "REQUEST",

            f"{request.method} {request.url.path}",

        )

        response = await call_next(request)

        elapsed = perf_counter() - start

        response.headers["X-Request-ID"] = request_id

        response.headers["X-Execution-Time"] = f"{elapsed:.4f}s"

        TZLogger.success(

            "REQUEST",

            f"{request.method} {request.url.path} ({elapsed:.3f}s)",

        )

        return response

class ResponseBuilder:

    @staticmethod
    def success(

        data=None,

        message="Sucesso.",

        status_code=200,

    ):

        return JSONResponse(

            status_code=status_code,

            content={

                "success": True,

                "message": message,

                "timestamp": datetime.utcnow().isoformat(),

                "data": data,

            },

        )

    @staticmethod
    def error(

        message,

        status_code=400,

        errors=None,

            status_code=status_code,

            content={

                "success": False,

                "message": message,

                "timestamp": datetime.utcnow().isoformat(),

                "errors": errors or [],

            },

        )

        self,

    # =====================================================
    # HEADERS
    # =====================================================

        headers = {
            "Content-Type": "application/json",
        }

            auth = ShopeeAuth(
                self.access_token,
            )

            headers.update(
                auth.get_authorization_header()
            )

        return headers

    # REST GET
    # =====================================================

    def get(
        self,
        endpoint: str,
        params: dict | None = None,
    ) -> dict[str, Any]:

        response = requests.get(

            endpoint,

            headers=self.get_headers(),

            params=params,

            timeout=self.DEFAULT_TIMEOUT,

        )

        response.raise_for_status()

        return response.json()

    # =====================================================
    # REST POST
    # =====================================================

    def post(
        self,
        endpoint: str,
        payload: dict | None = None,
    ) -> dict[str, Any]:

        response = requests.post(

            endpoint,

            headers=self.get_headers(),

            json=payload or {},

            timeout=self.DEFAULT_TIMEOUT,

        )

        response.raise_for_status()

        return response.json()

    # =====================================================
    # GRAPHQL
    # =====================================================

    def execute(
        self,
        *,
        query: str,
        variables: dict | None = None,
    ) -> dict[str, Any]:

        payload = {

            "query": query,

            "variables": variables or {},

        }

        response = requests.post(

            settings.SHOPEE_GRAPHQL_URL,

            json=payload,

            headers=self.get_headers(),

            timeout=self.DEFAULT_TIMEOUT,

        )

        response.raise_for_status()

from collections import deque
from datetime import datetime, timedelta

from app.integrations.shopee.constants import (
    RATE_LIMIT_PER_HOUR,
)


class RateLimiter:

    def __init__(
        self,
        limit: int = RATE_LIMIT_PER_HOUR,
    ):

        self.limit = limit
        self.requests = deque()

    def acquire(self):

        now = datetime.utcnow()

        while (
            self.requests
            and self.requests[0]
            < now - timedelta(hours=1)
        ):
            self.requests.popleft()

        if len(self.requests) >= self.limit:

            raise RuntimeError(
                "Shopee Rate Limit excedido."
            )

        self.requests.append(now)

Decision Engine V2

ResponsÃ¡vel por tomar decisÃµes

"""

from app.core.constants import Decision

from app.services.base_service import BaseService


class DecisionEngine(BaseService):

    SERVICE_NAME = "DECISION_ENGINE"

    PROMOTE_SCORE = 80

    TEST_SCORE = 60

    OBSERVE_SCORE = 40

    MIN_RATING = 4.5

    MIN_COMMISSION = 8

    MIN_SALES = 500

    MAX_PRICE = 500

    @classmethod
    def classify(cls, product: dict):

        cls.log_info("Analisando produto...")

        score = product["score"]

        if cls._can_promote(product):

            decision = Decision.PROMOTE.value

        elif score >= cls.TEST_SCORE:

            decision = Decision.TEST.value

        elif score >= cls.OBSERVE_SCORE:

            decision = Decision.OBSERVE.value

        else:

            decision = Decision.DISCARD.value

        cls.log_success(

            f"DecisÃ£o: {decision}"

        )

        return decision

    # =====================================================
    # REGRAS
    # =====================================================

    @classmethod
    def _can_promote(cls, product):

        if product["score"] < cls.PROMOTE_SCORE:

            return False

        if product["commission"] < cls.MIN_COMMISSION:

            return False

        if product["rating"] < cls.MIN_RATING:

            return False

        if product["sales"] < cls.MIN_SALES:

            return False

        if product["price"] > cls.MAX_PRICE:

            return False

        return True

class RankingEngine(BaseService):

    @classmethod
    def rank(cls, products: list[dict]) -> list[dict]:

        cls.log_info(
            "Gerando ranking..."
        )

        ranking = sorted(

            products,

            key=lambda product: (

                product["score"],

                product["sales"],

                product["commission"],

                product["rating"],

            ),

            reverse=True,

        )

        cls.log_success(

            f"{len(ranking)} produtos classificados."

        )

        return ranking

    @classmethod
    def top(cls, products, limit=10):

        return cls.rank(products)[:limit]

Pipeline oficial de prospecÃ§Ã£o

from sqlalchemy.orm import Session

from app.integrations.shopee.services.product_service import (
    ShopeeProductService,
)

from app.mappers.trend_product_mapper import (
    TrendProductMapper,
)

from app.services.product_validator import (
    ProductValidator,
)

from app.services.product_score import (
    ProductScore,
)

from app.services.decision_engine import (
    DecisionEngine,
)

from app.services.ranking_engine import (
    RankingEngine,
)

from app.services.persistence_service import (
    PersistenceService,
)

from app.services.executive_report import (
    ExecutiveReport,
)


class ScoutPipeline:

    def __init__(

        self,

        db: Session,

    ):

        self.db = db

        self.product_service = ShopeeProductService()

    # =====================================================
    # RUN
    # =====================================================

    def run(

        self,

        keyword: str = "",

        page: int = 1,

        limit: int = 20,

    ):

        dto_products = self.product_service.search_products(

            keyword=keyword,

            page=page,

            limit=limit,

        )

        products = [

            TrendProductMapper.from_shopee(dto)

            for dto in dto_products

        ]

        for product in products:

            ProductValidator.validate(

                {

                    "item_id": product.item_id,

                    "shop_id": product.shop_id,

                    "name": product.name,

                    "price": product.price,

                    "commission": product.commission,

                    "rating": product.rating,

                    "sales": product.sales,

                }

            )

            product.score = ProductScore.calculate(

                commission=product.commission,

                rating=product.rating,

                sales=product.sales,

            )

            product.decision = DecisionEngine.classify(

                {

                    "score": product.score,

                }

            )

        RankingEngine.rank(products)

        PersistenceService.save_products(

            self.db,

            products,

        )

        return [

            ExecutiveReport.build(

                {

                    "name": product.name,

                    "score": product.score,

                    "decision": product.decision,

                    "commission": product.commission,

                    "rating": product.rating,

                    "sales": product.sales,

                    "price": product.price,

                }

            )

            for product in products

        ]

"""

================================

Base Repository.

Centraliza todas as operaÃ§Ãµes CRUD.

Todos os Repositories da aplicaÃ§Ã£o
devem herdar desta classe.
"""

from typing import Any, Generic, TypeVar

from sqlalchemy import func, select
from sqlalchemy.orm import Session

from app.database.base import BaseModel

ModelType = TypeVar("ModelType", bound=BaseModel)


class BaseRepository(Generic[ModelType]):

    model = None

    # =====================================================
    # CREATE
    # =====================================================

    @classmethod
    def create(
        cls,
        db: Session,
        obj: ModelType,
    ) -> ModelType:

        db.add(obj)
        db.commit()
        db.refresh(obj)

        return obj

    # =====================================================
    # UPDATE
    # =====================================================

    @classmethod
    def update(
        cls,
        db: Session,
        obj: ModelType,
    ) -> ModelType:

        db.commit()
        db.refresh(obj)

        return obj

    # =====================================================
    # DELETE
    # =====================================================

    @classmethod
    def delete(
        cls,
        db: Session,
        obj: ModelType,
    ) -> None:

        db.delete(obj)
        db.commit()

    # =====================================================
    # SOFT DELETE
    # =====================================================

    @classmethod
    def soft_delete(
        cls,
        db: Session,
        obj: ModelType,
    ) -> None:

        obj.is_active = False

        db.commit()
        db.refresh(obj)

    # =====================================================
    # RESTORE
    # =====================================================

    @classmethod
    def restore(
        cls,
        db: Session,
        obj: ModelType,
    ) -> None:

        obj.is_active = True

        db.commit()
        db.refresh(obj)

    # =====================================================
    # GET ID
    # =====================================================

    @classmethod
    def get_by_id(
        cls,
        db: Session,
        id: int,
    ) -> ModelType | None:

        stmt = (
            select(cls.model)
            .where(
                cls.model.id == id,
                cls.model.is_active.is_(True),
            )
        )

        return db.execute(stmt).scalar_one_or_none()

    # =====================================================
    # GET UUID
    # =====================================================

    @classmethod
    def get_by_uuid(
        cls,
        db: Session,
        uuid: str,
    ) -> ModelType | None:

        stmt = (
            select(cls.model)
            .where(
                cls.model.uuid == uuid,
                cls.model.is_active.is_(True),
            )
        )

        return db.execute(stmt).scalar_one_or_none()

    # =====================================================
    # GET ALL
    # =====================================================

    @classmethod
    def get_all(
        cls,
        db: Session,
    ) -> list[ModelType]:

        stmt = (
            select(cls.model)
            .where(
                cls.model.is_active.is_(True)
            )
        )

        return list(db.execute(stmt).scalars().all())

    # =====================================================
    # GET MULTI
    # =====================================================

    @classmethod
    def get_multi(
        cls,
        db: Session,
        **filters: Any,
    ) -> list[ModelType]:

        stmt = select(cls.model)

        for field, value in filters.items():

            if hasattr(cls.model, field):

                stmt = stmt.where(
                    getattr(cls.model, field) == value
                )

        stmt = stmt.where(
            cls.model.is_active.is_(True)
        )

        return list(
            db.execute(stmt).scalars().all()
        )

    # =====================================================
    # EXISTS
    # =====================================================

    @classmethod
    def exists(
        cls,
        db: Session,
        **filters: Any,
    ) -> bool:

        stmt = select(cls.model)

        for field, value in filters.items():

            if hasattr(cls.model, field):

                stmt = stmt.where(
                    getattr(cls.model, field) == value
                )

        stmt = stmt.limit(1)

        return db.execute(stmt).scalar_one_or_none() is not None

    # =====================================================
    # PAGINATION
    # =====================================================

    @classmethod
    def paginate(
        cls,
        db: Session,
        page: int = 1,
        page_size: int = 20,
    ) -> list[ModelType]:

        stmt = (
            select(cls.model)
            .where(
                cls.model.is_active.is_(True)
            )
            .offset((page - 1) * page_size)
            .limit(page_size)
        )

        return list(
            db.execute(stmt).scalars().all()
        )

    # =====================================================
    # COUNT
    # =====================================================

    @classmethod
    def count(
        cls,
        db: Session,
    ) -> int:

        stmt = (
            select(
                func.count(cls.model.id)
            )
            .where(
                cls.model.is_active.is_(True)
            )
        )

        return db.scalar(stmt) or 0

            select(Product)
            .where(
                Product.name.ilike(f"%{text}%"),

            )

        )

    """

    # =====================================================
    # PESOS
    # =====================================================

    COMMISSION_WEIGHT = 0.50

    RATING_WEIGHT = 10

    SALES_DIVISOR = 1000

    SALES_LIMIT = 100

    # =====================================================
    # SCORE
    # =====================================================

    @classmethod
    def calculate(
        cls,
        *,
        commission: float,
        rating: float,
        sales: int,
    ) -> float:

        cls.log_info(
            "Calculando Score..."
        )

        commission_score = cls._commission_score(
            commission
        )

        rating_score = cls._rating_score(
            rating
        )

        sales_score = cls._sales_score(
            sales
        )

        final_score = (

            commission_score

            + rating_score

            + sales_score

        )

        final_score = round(
            final_score,
            2,
        )

        cls.log_success(
            f"Score calculado: {final_score}"
        )

        return final_score

        cls,

    # =====================================================
    # RATING
    # =====================================================

    @classmethod
    def _rating_score(
        cls,
        rating: float,
    ) -> float:

        return rating * cls.RATING_WEIGHT

    # =====================================================
    # SALES
    # =====================================================

    @classmethod
    def _sales_score(
        cls,
        sales: int,
    ) -> float:

        return min(

            sales / cls.SALES_DIVISOR,

            cls.SALES_LIMIT,

        )

            "success": (

                self.executed_tasks

            ),

import time

from app.core.logger import TZLogger


class RetryPolicy:

    def __init__(

        self,

        retries: int = 3,

        delay: int = 2,

    ):

        self.retries = retries

        self.delay = delay

    def execute(self, function, *args, **kwargs):

        last_exception = None

        for attempt in range(

            1,

            self.retries + 1,

        ):

            try:

                return function(

                    *args,

                    **kwargs,

                )

            except Exception as e:

                last_exception = e

                TZLogger.warning(

                    "RETRY",

                    f"Tentativa {attempt}/{self.retries} falhou."

                )

                if attempt < self.retries:

                    time.sleep(self.delay)

        TZLogger.error(

            "RETRY",

            "NÃºmero mÃ¡ximo de tentativas atingido."

        )

        raise last_exception

        self.tasks.append(task_id)

        return self

    def clear(self):

        self.tasks.clear()

    def run(self, context: TaskContext | None = None):

        if context is None:

        results = []

        TZLogger.info(

        )

        for task_id in self.tasks:

            task = TaskFactory.create(task_id)

            result = task.run(context)

            results.append(result)

            context.add_history(

                f"{task_id}: {result.message}"

            )

            if result.failed:

                TZLogger.error(

                    "TASK_PIPELINE",

                    f"Pipeline interrompido na Task {task_id}"

                )

                break

        TZLogger.success(

            "TASK_PIPELINE",

            "Pipeline finalizado."

        )

        return results

from dataclasses import dataclass, field

from typing import Any


@dataclass
class TaskResult:

    success: bool

    task: str

    message: str

    data: Any = None

    errors: list[str] = field(default_factory=list)

    warnings: list[str] = field(default_factory=list)

    metadata: dict = field(default_factory=dict)

    def add_error(self, message: str):

        self.errors.append(message)

    def add_warning(self, message: str):

        self.warnings.append(message)

    def add_metadata(self, key: str, value):

        self.metadata[key] = value

    @property
    def failed(self):

        return not self.success
